cmake_minimum_required(VERSION 3.21)
project(gpt2_c LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_executable(gpt2_c src/main.c src/model_config.c)
target_include_directories(gpt2_c PRIVATE include)
add_executable(gpt2_architecture_lesson src/architecture_lesson.c src/model_config.c)
target_include_directories(gpt2_architecture_lesson PRIVATE include)
add_executable(gpt2_batching_lesson src/batching_lesson.c)
add_executable(gpt2_embeddings_lesson src/embeddings_lesson.c)
add_executable(gpt2_attention_lesson src/attention_lesson.c)
add_executable(gpt2_block_lesson src/block_lesson.c)
if(NOT MSVC)
    target_link_libraries(gpt2_attention_lesson PRIVATE m)
    target_link_libraries(gpt2_block_lesson PRIVATE m)
endif()

set(TIKTOKEN_C_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/tiktoken-c-v0.9.1")
find_package(tiktoken-c 0.9.1 EXACT CONFIG QUIET PATHS "${TIKTOKEN_C_ROOT}" NO_DEFAULT_PATH)
if(TARGET tiktoken-c::shared)
    add_executable(gpt2_tokenizer_lesson src/tokenizer_lesson.c)
    target_link_libraries(gpt2_tokenizer_lesson PRIVATE tiktoken-c::shared)
    add_custom_command(TARGET gpt2_tokenizer_lesson POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:tiktoken-c::shared>"
            "$<TARGET_FILE_DIR:gpt2_tokenizer_lesson>")
else()
    message(STATUS "Optional tiktoken-c v0.9.1 not installed; C Lesson 2 target is skipped.")
endif()

if(MSVC)
    target_compile_options(gpt2_c PRIVATE /W4)
    target_compile_options(gpt2_architecture_lesson PRIVATE /W4)
    target_compile_options(gpt2_batching_lesson PRIVATE /W4)
    target_compile_options(gpt2_embeddings_lesson PRIVATE /W4)
    target_compile_options(gpt2_attention_lesson PRIVATE /W4)
    target_compile_options(gpt2_block_lesson PRIVATE /W4)
    if(TARGET gpt2_tokenizer_lesson)
        target_compile_options(gpt2_tokenizer_lesson PRIVATE /W4)
    endif()
else()
    target_compile_options(gpt2_c PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gpt2_architecture_lesson PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gpt2_batching_lesson PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gpt2_embeddings_lesson PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gpt2_attention_lesson PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(gpt2_block_lesson PRIVATE -Wall -Wextra -Wpedantic)
    if(TARGET gpt2_tokenizer_lesson)
        target_compile_options(gpt2_tokenizer_lesson PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
